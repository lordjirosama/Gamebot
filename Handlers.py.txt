# handlers.py
import random
import datetime
from telegram import Update, InlineKeyboardButton, InlineKeyboardMarkup
from telegram.ext import ContextTypes
from config import FOOTER, CHANNEL_URL, CHANNEL_NAME
from database import get_user, update_user, get_top_users

async def start_cmd(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_info = update.effective_user
    get_user(user_info.id, user_info.username, user_info.first_name)
    
    text = (
        f"⚔️ **ᴡᴇʟᴄᴏᴍᴇ ᴛᴏ ꜱᴏʟᴜʀɪx ᴄᴏᴍʙᴀᴛ ʀᴘɢ!** ⚔️\n\n"
        f"Hello **{user_info.first_name}**! Welcome to the ultimate action game.\n\n"
        f"🎮 **ǫᴜɪᴄᴋ ᴄᴏᴍᴍᴀɴᴅꜱ:**\n"
        f"• `/kill` — Fight monsters for EXP & Points\n"
        f"• `/duel @user` — Challenge another member\n"
        f"• `/profile` — View character stats\n"
        f"• `/shield` — Buy protection shield\n"
        f"• `/top` — Leaderboard"
        f"{FOOTER}"
    )
    keyboard = [[InlineKeyboardButton(f"📢 {CHANNEL_NAME}", url=CHANNEL_URL)]]
    await update.message.reply_text(text, parse_mode='Markdown', disable_web_page_preview=True, reply_markup=InlineKeyboardMarkup(keyboard))

async def kill_cmd(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_info = update.effective_user
    user = get_user(user_info.id, user_info.username, user_info.first_name)
    
    now = datetime.datetime.now()
    if user[12]:
        last_kill_time = datetime.datetime.fromisoformat(user[12])
        if (now - last_kill_time).total_seconds() < 30:
            remaining = 30 - int((now - last_kill_time).total_seconds())
            await update.message.reply_text(f"⌛ **ᴄᴏᴏʟᴅᴏᴡɴ!** Please wait **{remaining} seconds**.{FOOTER}", parse_mode='Markdown', disable_web_page_preview=True)
            return

    monsters = [("🐺 Shadow Wolf", 15, 25, 10), ("Goblin Warrior", 25, 40, 20), ("🧟 Undead Berserker", 40, 60, 35)]
    monster_name, pts_reward, coins_reward, exp_reward = random.choice(monsters)
    
    new_points = user[3] + pts_reward
    new_coins = user[4] + coins_reward
    new_exp = user[6] + exp_reward
    new_lvl = user[5]
    
    lvl_up_text = ""
    if new_exp >= new_lvl * 100:
        new_lvl += 1
        new_exp = 0
        lvl_up_text = f"\n\n🎊 **ʟᴇᴠᴇʟ ᴜᴘ!** You are now **Level {new_lvl}**!"
        
    update_user(user_info.id, points=new_points, coins=new_coins, exp=new_exp, level=new_lvl, last_kill=now.isoformat())
    
    text = (
        f"🗡️ **ʙᴀᴛᴛʟᴇ ᴠɪᴄᴛᴏʀʏ!**\n\n"
        f"You defeated a **{monster_name}**!\n"
        f"🎁 **Rewards:** +{pts_reward} Points 🎯 | +{coins_reward} Coins 🪙 | +{exp_reward} EXP ⭐"
        f"{lvl_up_text}{FOOTER}"
    )
    await update.message.reply_text(text, parse_mode='Markdown', disable_web_page_preview=True)

async def duel_cmd(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_info = update.effective_user
    attacker = get_user(user_info.id, user_info.username, user_info.first_name)
    
    target_user = update.message.reply_to_message.from_user if update.message.reply_to_message else None
    if not target_user:
        await update.message.reply_text(f"⚠️ **ᴜꜱᴀɢᴇ:** Reply to a user's message to duel them.{FOOTER}", parse_mode='Markdown', disable_web_page_preview=True)
        return

    defender = get_user(target_user.id, target_user.username, target_user.first_name)

    if defender[11]:
        shield_until = datetime.datetime.fromisoformat(defender[11])
        if datetime.datetime.now() < shield_until:
            await update.message.reply_text(f"🛡️ **ᴘʀᴏᴛᴇᴄᴛᴇᴅ!** **{target_user.first_name}** is protected by a **Solurix Shield**!{FOOTER}", parse_mode='Markdown', disable_web_page_preview=True)
            return

    stolen_points = random.randint(15, 35)
    if attacker[5] + random.randint(1, 10) >= defender[5] + random.randint(1, 10):
        update_user(attacker[0], points=attacker[3] + stolen_points)
        update_user(defender[0], points=max(0, defender[3] - stolen_points))
        text = f"⚔️ **ᴅᴜᴇʟ ᴠɪᴄᴛᴏʀʏ!**\n\n🥊 **{user_info.first_name}** won and stole **{stolen_points} Points** from **{target_user.first_name}**!{FOOTER}"
    else:
        update_user(defender[0], points=defender[3] + stolen_points)
        update_user(attacker[0], points=max(0, attacker[3] - stolen_points))
        text = f"🛡️ **ᴅᴜᴇʟ ᴅᴇꜰᴇᴀᴛ!**\n\n**{target_user.first_name}** defended successfully and took **{stolen_points} Points**!{FOOTER}"

    await update.message.reply_text(text, parse_mode='Markdown', disable_web_page_preview=True)

async def profile_cmd(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_info = update.effective_user
    user = get_user(user_info.id, user_info.username, user_info.first_name)
    
    shield_status = "❌ Inactive"
    if user[11]:
        shield_until = datetime.datetime.fromisoformat(user[11])
        if datetime.datetime.now() < shield_until:
            rem = shield_until - datetime.datetime.now()
            hours, remainder = divmod(int(rem.total_seconds()), 3600)
            minutes, _ = divmod(remainder, 60)
            shield_status = f"🛡️ Active ({hours}h {minutes}m left)"

    text = (
        f"👤 **ᴘʟᴀʏᴇʀ ᴘʀᴏꜰɪʟᴇ** — **{user[2]}**\n\n"
        f"⭐ **Level:** {user[5]} | **EXP:** {user[6]}/{user[5]*100}\n"
        f"🎯 **Points:** {user[3]} | 🪙 **Coins:** {user[4]}\n"
        f"🛡️ **Shield Status:** {shield_status}"
        f"{FOOTER}"
    )
    await update.message.reply_text(text, parse_mode='Markdown', disable_web_page_preview=True)

async def shield_cmd(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_info = update.effective_user
    user = get_user(user_info.id, user_info.username, user_info.first_name)

    text = (

        f"Your Points: **{user[3]}**\n\n"
        f"1️⃣ **1-Hour Shield** — 50 Points\n"
        f"2️⃣ **12-Hour Shield** — 300 Points\n"
        f"3️⃣ **24-Hour Shield** — 500 Points"
        f"{FOOTER}"
    )
    keyboard = [
        [InlineKeyboardButton("🛡️ Buy 1-Hour (50 Pts)", callback_data="buy_shield_1")],
        [InlineKeyboardButton("🛡️ Buy 12-Hour (300 Pts)", callback_data="buy_shield_12")],
        [InlineKeyboardButton("🛡️ Buy 24-Hour (500 Pts)", callback_data="buy_shield_24")]
    ]
    await update.message.reply_text(text, parse_mode='Markdown', disable_web_page_preview=True, reply_markup=InlineKeyboardMarkup(keyboard))

async def shield_callback(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    user_info = query.from_user
    user = get_user(user_info.id, user_info.username, user_info.first_name)

    costs = {"buy_shield_1": (50, 1), "buy_shield_12": (300, 12), "buy_shield_24": (500, 24)}
    if query.data in costs:
        cost, hours = costs[query.data]
        if user[3] < cost:
            await query.edit_message_text(f"❌ **ɪɴꜱᴜꜰꜰɪᴄɪᴇɴᴛ ᴘᴏɪɴᴛꜱ!** You need **{cost} Points**.{FOOTER}", parse_mode='Markdown', disable_web_page_preview=True)
            return

        now = datetime.datetime.now()
        current_shield = datetime.datetime.fromisoformat(user[11]) if user[11] and datetime.datetime.fromisoformat(user[11]) > now else now
        new_shield_time = current_shield + datetime.timedelta(hours=hours)

        update_user(user_info.id, points=user[3] - cost, shield_until=new_shield_time.isoformat())
        await query.edit_message_text(f"✅ **ꜱʜɪᴇʟᴅ ᴀᴄᴛɪᴠᴀᴛᴇᴅ!** Active for **{hours} Hours**.{FOOTER}", parse_mode='Markdown', disable_web_page_preview=True)

async def top_cmd(update: Update, context: ContextTypes.DEFAULT_TYPE):
    top_users = get_top_users()
    text = "🏆 **ɢʟᴏʙᴀʟ ꜱᴏʟᴜʀɪx ʟᴇᴀᴅᴇʀʙᴏᴀʀᴅ** 🏆\n\n"
    for i, u in enumerate(top_users, 1):
        medal = "🥇" if i == 1 else "🥈" if i == 2 else "🥉" if i == 3 else "👤"
        text += f"{medal} **{i}. {u[0]}** — {u[1]} Points (Lvl {u[2]})\n"
    text += FOOTER
    await update.message.reply_text(text, parse_mode='Markdown', disable_web_page_preview=True)

from config import OWNER_ID

async def broadcast_cmd(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.effective_user.id
    
    # Check if the user is the Owner
    if user_id != OWNER_ID:
        await update.message.reply_text("❌ **Access Denied!** Ye command sirf Bot Owner chala sakta hai.")
        return

    if not context.args:
        await update.message.reply_text("⚠️ **Usage:** `/broadcast Your Message Here`", parse_mode='Markdown')
        return

    message = " ".join(context.args)
    await update.message.reply_text(f"📢 **Broadcast Sent:**\n\n{message}", parse_mode='Markdown')
